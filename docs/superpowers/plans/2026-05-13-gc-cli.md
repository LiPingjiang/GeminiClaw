> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

```typescript
static loadSkills(skillsDir: string): SkillMeta[] {
  try {
    if (!this.existsAndIsDir(skillsDir)) {
      console.warn(`Skills directory does not exist: ${skillsDir}`)
      return []
    }

    const skills: SkillMeta[] = []
    const entries = readdirSync(skillsDir)

    for (const entry of entries) {
      if (entry === ".registry.json" || entry.startsWith(".")) continue

      const entryPath = join(skillsDir, entry)
      const stat = statSync(entryPath)

      if (stat.isDirectory()) {
        // 子目录格式：<name>/SKILL.md
        const skillFile = join(entryPath, "SKILL.md")
        try {
          statSync(skillFile)
          const skill = this.parseSkillFile(skillFile)
          if (skill) skills.push({ ...skill, name: entry })
        } catch {
          // 目录下没有 SKILL.md，跳过
        }
      } else if (this.isSkillFile(entry)) {
        // 旧格式：<name>_SKILL.md
        const skill = this.parseSkillFile(entryPath)
        if (skill) skills.push(skill)
      }
    }

    return skills
  } catch (error) {
    console.error(`Failed to load skills from ${skillsDir}:`, error)
    return []
  }
}
```

创建 `src/cli/lib/skill-registry.ts`：

```typescript
import { readFileSync, writeFileSync, existsSync, mkdirSync } from "fs"
import { join } from "path"

export interface SkillRegistryEntry {
  source: string           // "github:obra/superpowers" | "mt-skillhub:<name>" | "local"
  installedAt: string      // ISO 8601
  version: string | null
  updateUrl: string | null
}

export interface SkillRegistryData {
  version: number
  skills: Record<string, SkillRegistryEntry>
}

const REGISTRY_FILE = ".registry.json"

export class SkillRegistry {
  private skillsDir: string
  private data: SkillRegistryData

  constructor(skillsDir: string) {
    this.skillsDir = skillsDir
    this.data = this.load()
  }

  private registryPath(): string {
    return join(this.skillsDir, REGISTRY_FILE)
  }

  private load(): SkillRegistryData {
    const path = this.registryPath()
    if (!existsSync(path)) {
      return { version: 1, skills: {} }
    }
    try {
      return JSON.parse(readFileSync(path, "utf-8")) as SkillRegistryData
    } catch {
      return { version: 1, skills: {} }
    }
  }

  private save(): void {
    mkdirSync(this.skillsDir, { recursive: true })
    writeFileSync(
      this.registryPath(),
      JSON.stringify(this.data, null, 2),
      "utf-8"
    )
  }

  register(
    name: string,
    opts: { source: string; version?: string; updateUrl?: string }
  ): void {
    this.data.skills[name] = {
      source: opts.source,
      installedAt: new Date().toISOString(),
      version: opts.version ?? null,
      updateUrl: opts.updateUrl ?? null,
    }
    this.save()
  }

  get(name: string): SkillRegistryEntry | undefined {
    return this.data.skills[name]
  }

  list(): Array<{ name: string } & SkillRegistryEntry> {
    return Object.entries(this.data.skills).map(([name, entry]) => ({
      name,
      ...entry,
    }))
  }

  groupBySource(): Record<string, Array<{ name: string } & SkillRegistryEntry>> {
    const groups: Record<string, Array<{ name: string } & SkillRegistryEntry>> = {}
    for (const item of this.list()) {
      if (!groups[item.source]) groups[item.source] = []
      groups[item.source].push(item)
    }
    return groups
  }
}
```

创建 `src/cli/lib/workspace.ts`：

```typescript
import { existsSync, readFileSync } from "fs"
import { join, resolve, dirname } from "path"
import yaml from "js-yaml"

export interface WorkspacePaths {
  root: string         // workspace 根目录绝对路径
  soulFile: string
  userFile: string
  identityFile: string
  memoryFile: string
  agentsFile: string
  toolsFile: string
  memoryDir: string
  skillsDir: string    // GeminiClaw skills 目录
}

/**
 * 从 config.yaml 解析 workspace 和 skills 路径
 * 查找顺序：
 *   1. 环境变量 GC_CONFIG（指定 config.yaml 路径）
 *   2. 当前目录 ./config.yaml
 *   3. GeminiClaw 项目根目录（向上查找 package.json name=geminiclaw）
 */
export function resolveWorkspace(): WorkspacePaths {
  const configPath = findConfigFile()
  const projectRoot = dirname(configPath)

  let workspaceDir = ".workspace"
  let skillsDir = "skills"

  try {
    const raw = yaml.load(readFileSync(configPath, "utf-8")) as any
    workspaceDir = raw?.workspace?.dir ?? ".workspace"
    skillsDir = raw?.skills?.dir ?? "skills"
  } catch {
    // 使用默认值
  }

  const root = resolve(projectRoot, workspaceDir)
  const skillsDirAbs = resolve(projectRoot, skillsDir)

  return {
    root,
    soulFile: join(root, "SOUL.md"),
    userFile: join(root, "USER.md"),
    identityFile: join(root, "IDENTITY.md"),
    memoryFile: join(root, "MEMORY.md"),
    agentsFile: join(root, "AGENTS.md"),
    toolsFile: join(root, "TOOLS.md"),
    memoryDir: join(root, "memory"),
    skillsDir: skillsDirAbs,
  }
}

function findConfigFile(): string {
  // 1. 环境变量
  if (process.env.GC_CONFIG) return resolve(process.env.GC_CONFIG)

  // 2. 当前目录
  const cwd = join(process.cwd(), "config.yaml")
  if (existsSync(cwd)) return cwd

  // 3. 向上查找含 package.json(name=geminiclaw) 的目录
  let dir = process.cwd()
  for (let i = 0; i < 8; i++) {
    const pkg = join(dir, "package.json")
    if (existsSync(pkg)) {
      try {
        const p = JSON.parse(readFileSync(pkg, "utf-8"))
        if (p.name === "geminiclaw") {
          const cfg = join(dir, "config.yaml")
          if (existsSync(cfg)) return cfg
        }
      } catch {}
    }
    const parent = dirname(dir)
    if (parent === dir) break
    dir = parent
  }

  throw new Error(
    "config.yaml not found. Set GC_CONFIG env var or run from GeminiClaw project directory."
  )
}
```

export type OutputFormat = "human" | "json"

export function printHuman(text: string): void {
  console.log(text)
}

export function printJson(data: unknown): void {
  console.log(JSON.stringify(data, null, 2))
}

export function printOutput(data: unknown, text: string, format: OutputFormat): void {
  if (format === "json") {
    printJson(data)
  } else {
    printHuman(text)
  }
}

export function printError(msg: string, format: OutputFormat): void {
  if (format === "json") {
    printJson({ error: msg })
  } else {
    console.error(`❌ ${msg}`)
  }
  process.exit(1)
}

}

import { Command } from "commander"
import { existsSync, readFileSync } from "fs"
import { join } from "path"
import { resolveWorkspace } from "../lib/workspace.js"
import { printOutput, printError, type OutputFormat } from "../lib/output.js"

export interface IdentityFiles {
  soul: string | null
  user: string | null
  identity: string | null
}

export function readIdentityFiles(workspaceRoot: string): IdentityFiles {
  const read = (filename: string): string | null => {
    const p = join(workspaceRoot, filename)
    return existsSync(p) ? readFileSync(p, "utf-8") : null
  }
  return {
    soul: read("SOUL.md"),
    user: read("USER.md"),
    identity: read("IDENTITY.md"),
  }
}

export function registerIdentityCommand(program: Command): void {
  const identity = program
    .command("identity")
    .description(
      `Identity management — who is this agent and who does it serve

WHAT

function redact(obj: unknown): unknown {
  if (typeof obj !== "object" || obj === null) return obj
  if (Array.isArray(obj)) return obj.map(redact)
  const result: Record<string, unknown> = {}
  for (const [k, v] of Object.entries(obj as Record<string, unknown>)) {
    if (["apiKey", "authToken", "clientSecret", "password", "token"].includes(k)) {
      result[k] = "***"
    } else {
      result[k] = redact(v)
    }
  }
  return result
}

OUTPUT

      }

    socket.on("connect", () => { socket.destroy(); resolve(true) })

    .map((f) => join(sourceWorkspace, f))
    .filter(existsSync)

  const memoryDir = join(sourceWorkspace, "memory")
  const dailyMemoryFiles: string[] = []
  if (existsSync(memoryDir)) {
    for (const f of readdirSync(memoryDir)) {
      if (f.match(/^\d{4}-\d{2}-\d{2}.*\.md$/)) {
        dailyMemoryFiles.push(join(memoryDir, f))
      }
    }
  }

  return { identityFiles, dailyMemoryFiles }
}

export function registerSyncCommand(program: Command): void {
  const sync = program
    .command("sync")
    .description(
      `Experience sync — import knowledge from OpenClaw or other agent workspaces

WHAT
  Copies identity files (SOUL.md, USER.md, IDENTITY.md, MEMORY.md, AGENTS.md, TOOLS.md),
  daily memory files, and skills from a source workspace into this GeminiClaw workspace.
  Skills are registered in .registry.json with source provenance.

WHEN
  Run once to bootstrap GeminiClaw with accumulated experience from another agent.
  This is a one-shot import — it does not set up ongoing sync.

NOTE
  TOOLS.md may contain OpenClaw-specific config. Review it after sync and adapt
  as needed for GeminiClaw's different architecture.`
    )

  sync
    .command("from <source>")
    .description("Import experience from a source workspace directory")
    .option("--skills-dir <path>", "Source skills directory (default: ~/.openclaw/skills)")
    .option("--dry-run", "Preview what would be copied without making changes")
    .action((source, opts) => {
      try {
        const ws = resolveWorkspace()
        const sourceAbs = source.replace(/^~/, process.env.HOME ?? "~")

        if (!existsSync(sourceAbs)) {
          printError(`Source workspace not found: ${sourceAbs}`, "human")
          return
        }

        const plan = collectSyncableFiles(sourceAbs)
        const skillsSource = (opts.skillsDir ?? "~/.openclaw/skills").replace(
          /^~/, process.env.HOME ?? "~"
        )

        // 收集 skills
        const skillDirs: string[] = []
        if (existsSync(skillsSource)) {
          for (const entry of readdirSync(skillsSource)) {
            const p = join(skillsSource, entry)
            if (statSync(p).isDirectory() && existsSync(join(p, "SKILL.md"))) {
              skillDirs.push(p)
            }
          }
        }

        console.log(`\n📋 Sync plan from: ${sourceAbs}`)
        console.log(`\n  Identity/memory files (${plan.identityFiles.length}):`)
        plan.identityFiles.forEach((f) => console.log(`    + ${basename(f)}`))
        console.log(`\n  Daily memory files (${plan.dailyMemoryFiles.length}):`)
        if (plan.dailyMemoryFiles.length > 0) {
          console.log(`    + ${plan.dailyMemoryFiles.length} files`)
        }
        console.log(`\n  Skills (${skillDirs.length}):`)
        skillDirs.forEach((d) => console.log(`    + ${basename(d)}`))

        if (opts.dryRun) {
          console.log(`\n⚡ Dry run — no changes made.`)
          return
        }

        // 执行同步
        mkdirSync(ws.root, { recursive: true })
        mkdirSync(join(ws.root, "memory"), { recursive: true })

        // 1. 复制 identity 文件
        for (const f of plan.identityFiles) {
          const dest = join(ws.root, basename(f))
          copyFileSync(f, dest)
          console.log(`  ✅ Copied ${basename(f)}`)
        }

        // 2. 复制每日记忆
        for (const f of plan.dailyMemoryFiles) {
          const dest = join(ws.root, "memory", basename(f))
          copyFileSync(f, dest)
        }
        if (plan.dailyMemoryFiles.length > 0) {
          console.log(`  ✅ Copied ${plan.dailyMemoryFiles.length} daily memory files`)
        }

        // 3. 安装 skills
        const registry = new SkillRegistry(ws.skillsDir)
        mkdirSync(ws.skillsDir, { recursive: true })
        let skillCount = 0
        for (const skillDir of skillDirs) {
          const name = basename(skillDir)
          const destDir = join(ws.skillsDir, name)
          mkdirSync(destDir, { recursive: true })
          // 复制 SKILL.md 和其他文件
          for (const f of readdirSync(skillDir)) {
            copyFileSync(join(skillDir, f), join(destDir, f))
          }
          registry.register(name, {
            source: "openclaw:local",
            updateUrl: null,
          })
          skillCount++
        }
        if (skillCount > 0) {
          console.log(`  ✅ Installed ${skillCount} skills`)
        }

        // 4. 生成适配提示
        const toolsMd = join(ws.root, "TOOLS.md")
        if (existsSync(toolsMd)) {
          writeFileSync(
            join(ws.root, "TOOLS-ADAPTATION-NOTES.md"),
            `# TOOLS.md Adaptation Notes\n\n` +
            `TOOLS.md was imported from OpenClaw workspace on ${new Date().toISOString().slice(0, 10)}.\n\n` +
            `## Items that need review for GeminiClaw:\n\n` +
            `- [ ] OpenClaw gateway port (18789) → GeminiClaw port (see config.yaml server.port)\n` +
            `- [ ] OpenClaw skill paths (~/.openclaw/skills/) → GeminiClaw skills/ directory\n` +
            `- [ ] sandboxai / mcli proxy config → review providers in config.yaml\n` +
            `- [ ] launchd plist references → not applicable in GeminiClaw\n` +
            `\nReview TOOLS.md and update references as needed.\n`
          )
          console.log(`  📝 Created TOOLS-ADAPTATION-NOTES.md — please review TOOLS.md for OpenClaw-specific config`)
        }

        console.log(`\n✅ Sync complete!`)
        console.log(`   Workspace: ${ws.root}`)
        console.log(`   Run \`gc identity show\` to verify identity files.`)
        console.log(`   Run \`gc skill list\` to see imported skills.`)
      } catch (e) {
        printError(String(e), "human")
      }
    })
}

```markdown
## `gc` CLI

GeminiClaw 提供 `gc` CLI 工具，供 agent 和运维人员管理身份、记忆、skill 和配置。

### 安装

```bash
pnpm link --global   # 开发模式
# 或
pnpm build && npm install -g .
```

### 命令概览

| 命令 | 说明 |
|---|---|
| `gc identity show` | 查看 agent 身份（SOUL.md / USER.md） |
| `gc memory show` | 查看长期记忆 |
| `gc memory daily [date]` | 查看每日记忆 |
| `gc memory append <text>` | 追加长期记忆 |
| `gc skill list` | 列出所有 skill（含来源） |
| `gc skill info <name>` | 查看 skill 详情 |
| `gc skill install <path>` | 安装 skill（记录溯源） |
| `gc skill sources` | 查看所有 skill 来源汇总 |
| `gc config show` | 查看配置（secrets 脱敏） |
| `gc status` | 查看运行状态 |
| `gc sync from <path>` | 从其他 agent workspace 导入经验 |

所有命令支持 `--json` 输出机器可读格式。

### Skill 溯源机制

GeminiClaw 的 skill 系统内置溯源追踪，这是相对于 OpenClaw、Hermes 等系统的差异化特性。

每个 skill 安装时都会在 `skills/.registry.json` 中记录：

```json
{
  "brainstorming": {
    "source": "github:obra/superpowers",
    "installedAt": "2026-05-13T09:00:00Z",
    "version": "v5.1.0",
    "updateUrl": "https://github.com/obra/superpowers/tree/main/skills/brainstorming"
  }
}
```

**source 格式约定：**
- `github:<owner>/<repo>` — 来自 GitHub 开源项目
- `mt-skillhub:<skill-name>` — 来自美团内部 SkillHub
- `openclaw:local` — 从 OpenClaw workspace 导入
- `local` — 本地手动创建

**意义：**
- 随时可以追溯每个 skill 的上游来源
- `updateUrl` 为未来自动更新检查预留接口
- `gc skill sources` 提供整体 skill 生态的来源视图
```
